const express = require('express'); const session = require('express-session'); const path = require('path'); const app = express(); app.use(express.json()); app.use(express.urlencoded({ extended: true })); app.use(express.static(path.join(__dirname, 'public'))); app.use(session({ secret: 'cineva-secret-key-12345', resave: false, saveUninitialized: false, cookie: { maxAge: 24 * 60 * 60 * 1000 } })); app.get('/', (req, res) => { if (req.session) { req.session.authenticated = false; } res.sendFile(path.join(__dirname, 'public', 'index.html')); }); app.get('/menu.html', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'menu.html')); }); app.get('/watch.html', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'watch.html')); }); app.get('/admin-login', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'admin-login.html')); }); function checkAdminAuth(req, res, next) { if (req.session && req.session.adminLoggedIn) { next(); } else { if (req.path.startsWith('/api/')) { return res.status(401).json({ ok: false, message: 'Unauthorized' }); } res.redirect('/admin-login'); } } app.get('/admin', checkAdminAuth, (req, res) => { res.sendFile(path.join(__dirname, 'public', 'admin.html')); }); app.post('/api/member/login', (req, res) => { const { key } = req.body; if (key && key.startsWith('VIP-')) { req.session.authenticated = true; return res.json({ ok: true }); } res.status(400).json({ ok: false, message: 'รหัส Key ไม่ถูกต้อง หรือหมดอายุ' }); }); app.get('/api/member/verify', (req, res) => { if (req.session && req.session.authenticated) { return res.json({ ok: true }); } res.status(401).json({ ok: false }); }); app.post('/api/member/logout', (req, res) => { if (req.session) { req.session.authenticated = false; } res.json({ ok: true }); }); app.post('/api/admin/login', (req, res) => { const { password } = req.body; if (password === 'admin1234') { req.session.adminLoggedIn = true; return res.json({ ok: true }); } res.status(400).json({ ok: false, message: 'รหัสผ่านแอดมินไม่ถูกต้อง' }); }); app.post('/api/admin/logout', (req, res) => { if (req.session) { req.session.adminLoggedIn = false; } res.json({ ok: true }); }); app.get('/api/admin/keys', checkAdminAuth, (req, res) => { res.json({ ok: true, keys: [] }); }); app.post('/api/admin/keys', checkAdminAuth, (req, res) => { res.json({ ok: true, message: 'สร้าง Key สำเร็จ' }); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
