const express = require('express'); const session = require('express-session'); const path = require('path'); const app = express(); // ตั้งค่า Middleware พื้นฐาน app.use(express.json()); app.use(express.urlencoded({ extended: true })); app.use(express.static(path.join(__dirname, 'public'))); // ตั้งค่า Session app.use(session({ secret: 'cineva-secret-key-12345', resave: false, saveUninitialized: false, cookie: { maxAge: 24 * 60 * 60 * 1000 } // 1 วัน })); // ================= หน้าเว็บทั่วไป (Frontend Routes) ================= // หน้าแรก (ให้กรอก Key ลูกค้าเสมอ) app.get('/', (req, res) => { if (req.session) { req.session.authenticated = false; } res.sendFile(path.join(__dirname, 'public', 'index.html')); }); // หน้าเมนูหลักหลังกรอก Key ถูก app.get('/menu.html', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'menu.html')); }); // หน้าดูหนัง app.get('/watch.html', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'watch.html')); }); // หน้า Login แอดมินแยกต่างหาก app.get('/admin-login', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'admin-login.html')); }); // ================= ระบบตรวจสอบสิทธิ์แอดมิน (Middleware) ================= function checkAdminAuth(req, res, next) { if (req.session && req.session.adminLoggedIn) { next(); } else { if (req.path.startsWith('/api/')) { return res.status(401).json({ ok: false, message: 'Unauthorized' }); } res.redirect('/admin-login'); } } // หน้าแดชบอร์ดแอดมิน app.get('/admin', checkAdminAuth, (req, res) => { res.sendFile(path.join(__dirname, 'public', 'admin.html')); }); // ================= API สำหรับจัดการระบบ (API Routes) ================= // API ตรวจสอบ Key ของลูกค้า app.post('/api/member/login', (req, res) => { const { key } = req.body; // กำหนดเงื่อนไขเช็ก Key ของคุณตรงนี้ (ตัวอย่างกำหนดให้ VIP-1234 เป็น Key ที่ถูกต้อง) if (key && key.startsWith('VIP-')) { req.session.authenticated = true; return res.json({ ok: true }); } res.status(400).json({ ok: false, message: 'รหัส Key ไม่ถูกต้อง หรือหมดอายุ' }); }); // API ตรวจสอบสถานะลูกค้า app.get('/api/member/verify', (req, res) => { if (req.session && req.session.authenticated) { return res.json({ ok: true }); } res.status(401).json({ ok: false }); }); // API ออกจากระบบลูกค้า app.post('/api/member/logout', (req, res) => { if (req.session) { req.session.authenticated = false; } res.json({ ok: true }); }); // API ล็อกอินเข้าหลังบ้านแอดมิน app.post('/api/admin/login', (req, res) => { const { password } = req.body; // ตั้งรหัสผ่านแอดมินของคุณตรงนี้ (เช่น admin1234) if (password === 'admin1234') { req.session.adminLoggedIn = true; return res.json({ ok: true }); } res.status(400).json({ ok: false, message: 'รหัสผ่านแอดมินไม่ถูกต้อง' }); }); // API ออกจากระบบแอดมิน app.post('/api/admin/logout', (req, res) => { if (req.session) { req.session.adminLoggedIn = false; } res.json({ ok: true }); }); // API สำหรับจัดการ Key ในหน้าแอดมิน (ตัวอย่าง Endpoint สร้าง Key) app.get('/api/admin/keys', checkAdminAuth, (req, res) => { // ส่งข้อมูล Key ทั้งหมดกลับไปแสดงผลที่หน้าแอดมิน res.json({ ok: true, keys: [] }); }); app.post('/api/admin/keys', checkAdminAuth, (req, res) => { // โค้ดสำหรับสร้าง Key ใหม่ลงฐานข้อมูล res.json({ ok: true, message: 'สร้าง Key สำเร็จ' }); }); // เริ่มรันเซิร์ฟเวอร์ const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
