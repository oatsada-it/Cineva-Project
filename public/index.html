const express = require('express'); const session = require('express-session'); const path = require('path'); const app = express(); app.use(express.json()); app.use(express.urlencoded({ extended: true })); app.use(express.static(path.join(__dirname, 'public'))); app.use(session({ secret: 'cineva-secret-key-12345', resave: false, saveUninitialized: false, cookie: { maxAge: 24 * 60 * 60 * 1000 } })); // 1. หน้าแรกสุด: หน้ากรอก Key ลูกค้า (บังคับหยุดตรงนี้เสมอ) app.get('/', (req, res) => { if (req.session) { req.session.authenticated = false; } res.sendFile(path.join(__dirname, 'public', 'index.html')); }); // 2. หน้าเมนูหนังหลังกรอก Key ถูกต้อง app.get('/menu.html', (req, res) => { if (req.session && req.session.authenticated) { return res.sendFile(path.join(__dirname, 'public', 'menu.html')); } res.redirect('/'); }); // 3. หน้าเล่นหนัง / ดูรายละเอียด app.get('/watch.html', (req, res) => { if (req.session && req.session.authenticated) { return res.sendFile(path.join(__dirname, 'public', 'watch.html')); } res.redirect('/'); }); // 4. หน้าใส่รหัสผ่านแอดมินหลังบ้าน app.get('/admin-login', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'admin-login.html')); }); // ระบบตรวจสอบสิทธิ์แอดมิน function checkAdminAuth(req, res, next) { if (req.session && req.session.adminLoggedIn) { next(); } else { if (req.path.startsWith('/api/')) { return res.status(401).json({ ok: false, message: 'Unauthorized' }); } res.redirect('/admin-login'); } } // 5. หน้าแดชบอร์ดแอดมิน (ตามภาพที่ 3) app.get('/admin', checkAdminAuth, (req, res) => { res.sendFile(path.join(__dirname, 'public', 'admin.html')); }); // --- API สำหรับจัดการระบบ --- app.post('/api/member/login', (req, res) => { const { key } = req.body; if (key && key.trim() !== '') { req.session.authenticated = true; return res.json({ ok: true }); } res.status(400).json({ ok: false, message: 'กรุณากรอกรหัส Key' }); }); app.post('/api/member/logout', (req, res) => { if (req.session) { req.session.authenticated = false; } res.json({ ok: true }); }); app.post('/api/admin/login', (req, res) => { const { password } = req.body; if (password === 'admin1234') { // เปลี่ยนรหัสผ่านแอดมินได้ตรงนี้ req.session.adminLoggedIn = true; return res.json({ ok: true }); } res.status(400).json({ ok: false, message: 'รหัสผ่านแอดมินไม่ถูกต้อง' }); }); app.post('/api/admin/logout', (req, res) => { if (req.session) { req.session.adminLoggedIn = false; } res.json({ ok: true }); }); app.get('/api/admin/keys', checkAdminAuth, (req, res) => { res.json({ ok: true, keys: [] }); }); app.post('/api/admin/keys', checkAdminAuth, (req, res) => { res.json({ ok: true, message: 'สร้าง Key สำเร็จ' }); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
